\begin{equation}\label{eq:loglikelihood}
\mathcal{L}
\equiv
\sum_{k,n} \ell_{kn} \ln \left[\mathbb{P}(U_{kn}^{i} > U_{k'n'}^{i} \ \forall k'n' \neq kn)\right] 
=
\sum_{k,n} \ell_{kn} \ln \left[
\frac
{{w}_{n}^\epsilon \left({r}_{k}^\alpha  \bar{\delta}_{kn}\right)^{-\epsilon}}
{\sum_{k',n'} {w}_{n'}^\epsilon \left({r}_{k'}^\alpha  \bar{\delta}_{k'n'}\right)^{-\epsilon}}
\right]
.\end{equation}
\begin{equation} \label{eqn:utility:beliefs}
\tilde{U}_{kn}^{i} = 
\underbrace{\epsilon \ln\left({\tilde{w}_n}/\left({\tilde{P}^{1-\alpha}\tilde{r}_{k}^{\alpha} \delta_{kn}}\right)\right)}_{\equiv \tilde{U}_{kn}} + \nu_{kn}^i
,
\end{equation}
\begin{equation}\label{eq:gran_loglikelihood}
\mathcal{L}
\equiv
\sum_{k,n} \ell_{kn} \ln \left[\mathbb{P}(\tilde{U}_{kn}^{i} > \tilde{U}_{k'n'}^{i} \ \forall k'n' \neq kn)\right]
=
\sum_{k,n} \ell_{kn} \ln \left[
\frac
{\tilde{w}_{n}^\epsilon \left(\tilde{r}_{k}^\alpha  \bar{\delta}_{kn}\right)^{-\epsilon}}
{\sum_{k',n'} \tilde{w}_{n'}^\epsilon \left(\tilde{r}_{k'}^\alpha  \bar{\delta}_{k'n'}\right)^{-\epsilon}}
\right]
.\end{equation}
\begin{equation}\label{eqn:cont_laborallocation}
\frac{\ell_{kn}}{L}
=
\frac
{w_{n}^\epsilon \left(r_{k}^\alpha  \delta_{kn}\right)^{-\epsilon}}
{\sum_{k',n'}w_{n'}^\epsilon \left(r_{k'}^\alpha  \delta_{k'n'}\right)^{-\epsilon}}
.
\end{equation}
\begin{align}
\hat{w}_{n}
&=
\hat{A}_n
\left(\sum_{k} \hat{y}_{kn}\frac{y_{kn}}{\sum_{k'} y_{k'n}}\right)^{\frac{1}{1-\sigma}}
\hat{P}
\hat{Y}^{\frac{1}{\sigma-1}}
\label{eqn:exacthatalgebra:wages}
\\
\hat{r}_{k}
&=
\hat{T}_{k}^{-1} \sum_{n}\hat{y}_{kn}\frac{y_{kn}}{\sum_{n'}y_{kn'}}
\label{eqn:exacthatalgebra:rents}
\\
\hat{\ell}_{kn} 
&=
\frac{\hat{w}_{n}^{\epsilon}\left(\hat{r}_{k}^{\alpha} \hat{\bar{\delta}}_{kn}\hat{ \lambda}_{kn} \right)^{-\epsilon}}{\sum_{k',n'} \hat{w}_{n'}^{\epsilon}\left(\hat{r}_{k'}^{\alpha} \hat{\bar{\delta}}_{k'n'}\hat{ \lambda}_{k'n'} \right)^{-\epsilon} \frac{\ell_{k'n'}}{L}} \qquad \text{ if } \ell_{kn} > 0
\label{eqn:exacthatalgebra:laborallocation}
.\end{align}

To simulate draws from a nested-logit distribution $F(\mathring{\nu}^i)$,
we first represent $\mathring{\nu}^i \sim F(\mathring{\nu}^i)$ using 
a standard Gumbel distribution $\mathcal{G}(0,1)$ and a positive stable distribution (PSD) $\mathcal{P}(\zeta)$,

Second, we apply \cite{Ridout:2009}’s method to simulate $Z_{B_z}^i \sim \mathcal{P}(\zeta)$ for $\zeta = 0.25$.

to construct the draws from the nested-logit distribution.
